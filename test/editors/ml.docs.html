File

File = item:Decl*

Decl

Decl = kind:(Import | TypeDecl | DataDecl | LetDecl)

Sig

Sig = item:SigItem*

SigItem

SigItem = kind:(TypeDecl | ValDecl)

ValDecl

ValDecl = kw:'val' name:ident colon:':' ty:Type semi:';'

Import

Import = kw:'import' path:Path semi:';'

Path

Path = head:ident rest:PathSeg*

PathSeg

PathSeg = dot:'.' name:ident

TypeDecl

TypeDecl = kw:'type' name:ident eq:'=' ty:Type semi:';'

TypeGroup

TypeGroup = '(' ty:Type ')'

DataDecl

DataDecl = kw:'data' name:ident body:DataBody

DataBody

DataBody = '{' ctor:Ctor* '}'

Ctor

Ctor = bar:'|' name:ident payload:CtorPayload?

CtorPayload

CtorPayload = '(' ty:Type* ')' sep ','

LetDecl

LetDecl =
  kw:'let'
  name:ident
  param:Param*
  colon:':'
  ty:Type
  eq:'='
  body:Expr
  semi:';'

Param

Param = kind:(PlainParam | LabelledParam)

PlainParam

PlainParam = lparen:'(' name:ident colon:':' ty:Type rparen:')'

LabelledParam

LabelledParam = tilde:'~' name:ident

List

List = '[' elem:Expr* ']' sep ',' trailing

If

If = kw:'if' cond:Expr then_kw:'then' t:Expr else_:Else?

Else

Else = kw:'else' e:Expr

Stmt

Stmt = e:Expr semi:';'

Label

Label = tilde:'~' name:ident colon:':'

Arg

Arg = label:Label? value:Expr

Type

Type = expression

  atoms ident TypeGroup
  infix '->' (1 right) '*' (10 left)

Expr

Expr = expression

  atoms int ident List If
  prefix '!' (60 right)
  infix
    '==' (5 left)
    '<' (5 left)
    '+' (10 left)
    '-' (10 left)
    '*' (20 left)
    '/' (20 left)
  postfix
    x'.'ident (100)
    x'('Arg sep ',' trailing-on-break')' (100)
    x'{'Stmt'}' (100)
