File

File = header:Module? item:Item*

Module

Module = kw:'module' path:Path semi:';'

DocBlock

DocBlock = line:doc_comment+

Item

Item = doc:DocBlock? def:Definition

Definition

Definition =
  kind:(
    Import
    | Def
    | ValDef
    | VarDef
    | TypeDef
    | Record
    | Interface
    | Effect
    | Namespace
  )

Path

Path = first:ident rest:PathSegment*

PathSegment

PathSegment = sep:'::' name:ident

Import

Import = kw:'import' path:Path semi:';'

Param

Param = name:ident ann:TypeAnn?

TypeAnn

TypeAnn = colon:':' ty:Type

Params

Params = '(' param:Param* ')' sep ',' trailing-on-break

Def

Def =
  kw:'def'
  name:ident
  params:Params?
  ret:TypeAnn?
  eq:'='
  body:Expr
  semi:';'

ValDef

ValDef = kw:'val' name:ident ann:TypeAnn? eq:'=' value:Expr semi:';'

VarDef

VarDef = kw:'var' name:ident ann:TypeAnn? eq:'=' value:Expr semi:';'

TypeDef

TypeDef = kw:'type' name:ident eq:'=' ty:Type semi:';'

Record

Record = kw:'record' name:ident fields:Params

Operation

Operation = doc:DocBlock? kw:'def' name:ident params:Params ret:TypeAnn?

InterfaceBody

InterfaceBody = '{' op:Operation* '}'

Interface

Interface = kw:'interface' name:ident body:InterfaceBody

Effect

Effect = kw:'effect' name:ident params:Params ret:TypeAnn? semi:';'

NamespaceBody

NamespaceBody = '{' item:Item* '}'

Namespace

Namespace = kw:'namespace' name:ident body:NamespaceBody

Type

Type = from:BoxedType to_:TypeArrow?

TypeArrow

TypeArrow = arrow:'->' result:BoxedType

BoxedType

BoxedType = base:AtomicType capture:CaptureSet?

CaptureSet

CaptureSet = kw:'at' set:CaptureList

CaptureList

CaptureList = '{' name:ident* '}' sep ','

AtomicType

AtomicType = name:ident args:TypeArgs?

TypeArgs

TypeArgs = '[' arg:Type+ ']' sep ','

Block

Block = '{' stmt:Stmt* '}'

Stmt

Stmt = kind:(ValDef | VarDef | Def | ExprStmt)

ExprStmt

ExprStmt = expr:Expr semi:';'

If

If = kw:'if' cond:Paren then_:Expr else_:Else?

Else

Else = kw:'else' body:Expr

While

While = kw:'while' cond:Paren body:Expr

Paren

Paren = '(' inner:Expr ')'

Do

Do = kw:'do' name:ident args:Args

Args

Args = '(' arg:Expr* ')' sep ',' trailing-on-break

Return

Return = kw:'return' value:Expr

Resume

Resume = kw:'resume' args:Args

Box

Box = kw:'box' value:Expr

Unbox

Unbox = kw:'unbox' value:Expr

Fn

Fn = kw:'fn' params:Params arrow:'=>' body:Expr

Try

Try = kw:'try' body:Block handler:Handler+

Handler

Handler = kw:'with' name:ident body:HandlerBody

HandlerBody

HandlerBody = '{' clause:Clause* '}'

Clause

Clause = kw:'def' name:ident params:Params eq:'=' body:Expr

Match

Match = kw:'match' scrutinee:Paren body:MatchBody

MatchBody

MatchBody = '{' arm:MatchArm* '}'

MatchArm

MatchArm =
  kw:'case'
  pat:Pattern
  extra:AltPattern*
  guard:Guard?
  arrow:'=>'
  body:Expr

AltPattern

AltPattern = bar:'|' pat:Pattern

Guard

Guard = kw:'and' cond:Expr

Pattern

Pattern = head:(ident | int | string) args:PatternArgs?

PatternArgs

PatternArgs = '(' arg:Pattern* ')' sep ','

Hole

Hole = '<{' stmt:Stmt* '}>'
