File

File = item:Item*

Item

Item = kind:(Struct | Enum | Trait | Fn)

Struct

Struct = kw:'struct' name:ident body:StructBody

StructBody

StructBody = '{' field:Field* '}' sep ',' trailing-on-break

Field

Field = name:ident colon:':' ty:Type

Enum

Enum = kw:'enum' name:ident body:EnumBody

EnumBody

EnumBody = '{' variant:Variant* '}' sep ',' trailing-on-break

Variant

Variant = name:ident payload:VariantPayload?

VariantPayload

VariantPayload = shape:(TuplePayload | StructPayload)

TuplePayload

TuplePayload = '(' ty:Type* ')' sep ','

StructPayload

StructPayload = '{' field:Field* '}' sep ',' trailing-on-break

Trait

Trait = kw:'trait' name:ident body:TraitBody

TraitBody

TraitBody = '{' method_:MethodSig* '}'

MethodSig

MethodSig =
  kw:'fn'
  name:ident
  params:ParamList
  arrow:'->'
  ret_ty:Type
  semi:';'

Fn

Fn = kw:'fn' name:ident params:ParamList arrow:'->' ret_ty:Type body:Block

ParamList

ParamList = '(' param:Param* ')' sep ',' trailing-on-break

Param

Param = name:ident colon:':' ty:Type

Type

Type = name:ident

Block

Block = '{' stmt:Stmt* '}'

Stmt

Stmt = kind:(Let | ExprStmt)

Let

Let = kw:'let' name:ident eq:'=' value:Expr semi:';'

ExprStmt

ExprStmt = expr:Expr semi:';'

Match

Match = kw:'match' scrutinee:Expr body:MatchBody

MatchBody

MatchBody = '{' arm:MatchArm* '}' sep ',' trailing-on-break

MatchArm

MatchArm = pat:(ident | int) fat_arrow:'=>' body:Expr
